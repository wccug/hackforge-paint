cmake_minimum_required(VERSION 3.5)
project(HackforgeColorPicker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
# Fallback option if Qt6 isn't present:
# find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

add_executable(hackforge_colorpicker
    main.cpp
    ColorPickerWindow.cpp
    ColorPickerWindow.h
)

target_link_libraries(hackforge_colorpicker PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
# Use Qt5::Core Qt5::Widgets if compiling against Qt5.
